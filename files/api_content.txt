Verify this signature in your endpoint to confirm the request came from SocialSphear.// Verify webhook signature (Node.js example)import crypto from &apos;crypto&apos;; function verifyWebhookSignature(payload, signature, secret) { const expected = crypto .createHmac(&apos;sha256&apos;, secret) .update(payload, &apos;utf8&apos;) .digest(&apos;hex&apos;); return crypto.timingSafeEqual( Buffer.from(signature), Buffer.from(&apos;sha256=&apos; + expected) );} app.post(&apos;/webhook&apos;, (req, res) =&gt; { const signature = req.headers[&apos;x-socialsphear-signature&apos;]; const isValid = verifyWebhookSignature( JSON.stringify(req.body), signature, process.env.WEBHOOK_SECRET ); if (!isValid) return res.status(401).send(&apos;Invalid signature&apos;); const { event, data } = req.body; // Handle event... res.status(200).send(&apos;OK&apos;);});9.
