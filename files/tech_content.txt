The ClerkGuard NestJS guard verifies the JWT, extracts the userId and orgId, and attaches them to the request context. All downstream service calls use these verified identities ΓÇö no trust is placed on client-supplied user IDs.// ClerkGuard ΓÇö verifies JWT and attaches user context@Injectable()export class ClerkGuard implements CanActivate { async canActivate(context: ExecutionContext): Promise&lt;boolean&gt; { const request = context.switchToHttp().getRequest(); const token = extractBearerToken(request); const session = await clerkClient.verifyToken(token); request.userId = session.sub; request.orgId = session.org_id; return true; }}4. Scheduling EngineCRITICALThe scheduling engine is the most important system in SocialSphear. A post that fails to publish ΓÇö or publishes late ΓÇö destroys user trust. Every design decision here prioritises reliability above all else.4.1 BullMQ Queue ArchitectureQueuePurposeConcurrencyRetry Strategypost:scheduleReceives new scheduled post jobs50No retry ΓÇö idempotent entry pointpost:publishExecutes API call to social platform20 per platform3 retries, exponential backoff (30s, 2m, 10m)post:notifySends success/failure notifications1002 retries, 5s delayanalytics:ingestFetches post metrics from platform APIs105 retries, 1h delaytoken:refreshRefreshes expiring OAuth tokens53 retries, 1m delaymedia:processTriggers Cloudinary transforms after upload302 retries, 10s delay4.2 Post Lifecycle State MachineDRAFT | vSCHEDULED --&gt; QUEUED --&gt; PUBLISHING --&gt; PUBLISHED | +--&gt; FAILED --&gt; RETRYING | | +--&gt; CANCELLED +--&gt; FAILED_PERMANENT4.3 Temporal WorkflowsWorkflowTriggerStepsScheduledPostWorkflowUser schedules a post1. Database Design5.1 PostgreSQL Schema (Core Tables)-- Users (managed primarily by Clerk, mirrored here for FK references)CREATE TABLE users ( id UUID PRIMARY KEY DEFAULT gen_random_uuid(), clerk_id TEXT UNIQUE NOT NULL, email TEXT UNIQUE NOT NULL, name TEXT, avatar_url TEXT, created_at TIMESTAMPTZ DEFAULT now()); -- Workspaces (one per brand or client)CREATE TABLE workspaces ( id UUID PRIMARY KEY DEFAULT gen_random_uuid(), name TEXT NOT NULL, slug TEXT UNIQUE NOT NULL, logo_url TEXT, owner_id UUID REFERENCES users(id), plan TEXT NOT NULL DEFAULT &apos;free&apos;, created_at TIMESTAMPTZ DEFAULT now()); -- Social Accounts (one row per connected account)CREATE TABLE social_accounts ( id UUID PRIMARY KEY DEFAULT gen_random_uuid(), workspace_id UUID REFERENCES workspaces(id) ON DELETE CASCADE, platform TEXT NOT NULL, platform_user_id TEXT NOT NULL, username TEXT, access_token BYTEA NOT NULL, -- AES-256 encrypted refresh_token BYTEA, -- AES-256 encrypted token_expires_at TIMESTAMPTZ, status TEXT DEFAULT &apos;active&apos;, created_at TIMESTAMPTZ DEFAULT now(), UNIQUE(workspace_id, platform, platform_user_id)); -- Posts (the content unit)CREATE TABLE posts ( id UUID PRIMARY KEY DEFAULT gen_random_uuid(), workspace_id UUID REFERENCES workspaces(id) ON DELETE CASCADE, created_by UUID REFERENCES users(id), content TEXT, media_urls JSONB DEFAULT &apos;[]&apos;, platform_overrides JSONB DEFAULT &apos;{}&apos;, scheduled_at TIMESTAMPTZ, status TEXT DEFAULT &apos;draft&apos;, approval_status TEXT DEFAULT &apos;none&apos;, created_at TIMESTAMPTZ DEFAULT now(), updated_at TIMESTAMPTZ DEFAULT now()); -- Post Jobs (one per platform per post)CREATE TABLE post_jobs ( id UUID PRIMARY KEY DEFAULT gen_random_uuid(), post_id UUID REFERENCES posts(id) ON DELETE CASCADE, social_account_id UUID REFERENCES social_accounts(id), platform TEXT NOT NULL, status TEXT DEFAULT &apos;pending&apos;, platform_post_id TEXT, platform_url TEXT, published_at TIMESTAMPTZ, error_message TEXT, retry_count INTEGER DEFAULT 0, created_at TIMESTAMPTZ DEFAULT now()); -- Team Members (workspace access)CREATE TABLE team_members ( id UUID PRIMARY KEY DEFAULT gen_random_uuid(), workspace_id UUID REFERENCES workspaces(id) ON DELETE CASCADE, user_id UUID REFERENCES users(id) ON DELETE CASCADE, role TEXT NOT NULL DEFAULT &apos;editor&apos;, invite_status TEXT DEFAULT &apos;pending&apos;, joined_at TIMESTAMPTZ, UNIQUE(workspace_id, user_id));5.2 ClickHouse Analytics Schema-- Post metrics ΓÇö one row per metric snapshotCREATE TABLE post_metrics ( post_job_id UUID, workspace_id UUID, platform LowCardinality(String), recorded_at DateTime, likes UInt32 DEFAULT 0, comments UInt32 DEFAULT 0, shares UInt32 DEFAULT 0, saves UInt32 DEFAULT 0, reach UInt32 DEFAULT 0, impressions UInt32 DEFAULT 0, engagement_rate Float32) ENGINE = MergeTree() PARTITION BY toYYYYMM(recorded_at) ORDER BY (workspace_id, platform, recorded_at); -- Account growth ΓÇö one row per daily snapshotCREATE TABLE account_snapshots ( social_account_id UUID, workspace_id UUID, platform LowCardinality(String), snapshot_date Date, follower_count UInt32, following_count UInt32, post_count UInt32) ENGINE = MergeTree() PARTITION BY toYYYYMM(snapshot_date) ORDER BY (workspace_id, platform, snapshot_date);6.
